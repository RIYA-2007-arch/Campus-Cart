import React, { useEffect, useState } from "react";
import axios from "axios";
import "./App.css";

const API = import.meta.env.VITE_API_URL || "http://127.0.0.1:5000/api";

function EyeIcon({ visible }) {
  return visible ? (
    <svg viewBox="0 0 24 24" width="21" height="21" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round">
      <path d="M3 3l18 18" />
      <path d="M10.6 10.6a2 2 0 002.8 2.8" />
      <path d="M9.9 5.2A10.8 10.8 0 0112 5c5.5 0 9 7 9 7a14 14 0 01-3 3.8" />
      <path d="M6.2 6.2C3.9 7.8 3 12 3 12s3.5 7 9 7a9.8 9.8 0 004-.9" />
    </svg>
  ) : (
    <svg viewBox="0 0 24 24" width="21" height="21" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round">
      <path d="M2.5 12s3.5-7 9.5-7 9.5 7 9.5 7-3.5 7-9.5 7-9.5-7-9.5-7z" />
      <circle cx="12" cy="12" r="3" />
    </svg>
  );
}

function PasswordField({
  label,
  value,
  onChange,
  placeholder,
  autoComplete,
  required = true,
}) {
  const [visible, setVisible] = useState(false);

  return (
    <div className="form-group">
      <label>{label}</label>
      <div className="password-input-wrap">
        <input
          type={visible ? "text" : "password"}
          placeholder={placeholder}
          value={value}
          onChange={onChange}
          autoComplete={autoComplete}
          required={required}
        />
        <button
          type="button"
          className="password-toggle"
          onClick={() => setVisible((previous) => !previous)}
          aria-label={visible ? "Hide password" : "Show password"}
          title={visible ? "Hide password" : "Show password"}
        >
          <EyeIcon visible={visible} />
        </button>
      </div>
    </div>
  );
}

function App() {
  const [showLogin, setShowLogin] = useState(false);
  const [showRegister, setShowRegister] = useState(false);
  const [showForgot, setShowForgot] = useState(false);

  const [loginEmail, setLoginEmail] = useState("");
  const [loginPassword, setLoginPassword] = useState("");

  const [registerName, setRegisterName] = useState("");
  const [registerEmail, setRegisterEmail] = useState("");
  const [registerBranch, setRegisterBranch] = useState("");
  const [registerSemester, setRegisterSemester] = useState("");
  const [registerPassword, setRegisterPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");

  const [otpCode, setOtpCode] = useState("");
  const [otpSent, setOtpSent] = useState(false);
  const [emailVerified, setEmailVerified] = useState(false);

  const [otpLoading, setOtpLoading] = useState(false);
  const [verifyLoading, setVerifyLoading] = useState(false);
  const [registerLoading, setRegisterLoading] = useState(false);
  const [loginLoading, setLoginLoading] = useState(false);
  const [forgotLoading, setForgotLoading] = useState(false);

  const [authMessage, setAuthMessage] = useState("");
  const [authError, setAuthError] = useState("");
  const [loggedInUser, setLoggedInUser] = useState(null);

  const [forgotEmail, setForgotEmail] = useState("");
  const [forgotOtp, setForgotOtp] = useState("");
  const [forgotOtpSent, setForgotOtpSent] = useState(false);
  const [forgotOtpVerified, setForgotOtpVerified] = useState(false);
  const [newPassword, setNewPassword] = useState("");
  const [confirmNewPassword, setConfirmNewPassword] = useState("");

  const [products, setProducts] = useState([]);
  const [productsLoading, setProductsLoading] = useState(true);
  const [productsError, setProductsError] = useState("");
  const [searchText, setSearchText] = useState("");

  const categories = [
    { icon: "B", title: "Books", description: "Second-hand academic books" },
    { icon: "S", title: "Study Material", description: "Notes and study resources" },
    { icon: "E", title: "Electronics", description: "Student-use electronics" },
    { icon: "C", title: "Calculators", description: "Scientific and academic calculators" },
    { icon: "CE", title: "College Essentials", description: "Useful campus products" },
    { icon: "H", title: "Hostel Essentials", description: "Everyday hostel items" },
    { icon: "SP", title: "Sports", description: "Sports and fitness items" },
    { icon: "O", title: "Other", description: "More student products" },
  ];

  useEffect(() => {
    const fetchProducts = async () => {
      try {
        const response = await axios.get(`${API}/products`);
        const data = Array.isArray(response.data) ? response.data : [];

        setProducts(
          data.map((product) => {
            const price = Number(product.price || 0);
            const original = Number(product.original_price || 0);
            const meta = [
              product.branch,
              product.semester ? `Semester ${product.semester}` : null,
              product.condition,
            ].filter(Boolean);

            return {
              ...product,
              name: product.product_name || product.name || "Student Product",
              displayPrice: `₹${price.toLocaleString("en-IN")}`,
              originalPrice: original > price
                ? `₹${original.toLocaleString("en-IN")}`
                : "",
              saving: original > price
                ? `Save ₹${(original - price).toLocaleString("en-IN")}`
                : "",
              meta: meta.join(" • ") || product.category || "Student Product",
              campus: product.college || "AKS University",
            };
          })
        );
        setProductsError("");
      } catch (error) {
        console.error("Products error:", error);
        setProductsError("Products could not be loaded. Please check that the backend is running.");
      } finally {
        setProductsLoading(false);
      }
    };

    fetchProducts();
  }, []);

  const clearMessages = () => {
    setAuthMessage("");
    setAuthError("");
  };

  const resetOtpState = () => {
    setOtpCode("");
    setOtpSent(false);
    setEmailVerified(false);
  };

  const openLogin = () => {
    setShowRegister(false);
    setShowForgot(false);
    setShowLogin(true);
    clearMessages();
  };

  const openRegister = () => {
    setShowLogin(false);
    setShowForgot(false);
    setShowRegister(true);
    clearMessages();
    resetOtpState();
  };

  const openForgot = () => {
    setShowLogin(false);
    setShowRegister(false);
    setShowForgot(true);
    setForgotEmail(loginEmail);
    setForgotOtp("");
    setForgotOtpSent(false);
    setForgotOtpVerified(false);
    setNewPassword("");
    setConfirmNewPassword("");
    clearMessages();
  };

  const closeModals = () => {
    setShowLogin(false);
    setShowRegister(false);
    setShowForgot(false);
    clearMessages();
  };

  const handleRegisterEmailChange = (event) => {
    setRegisterEmail(event.target.value);
    resetOtpState();
    clearMessages();
  };

  const handleSendOtp = async () => {
    clearMessages();
    const email = registerEmail.trim().toLowerCase();

    if (!email) {
      setAuthError("Please enter your email address first.");
      return;
    }

    setOtpLoading(true);
    try {
      const response = await axios.post(`${API}/send-otp`, { email });
      setOtpSent(true);
      setEmailVerified(false);
      setOtpCode("");
      setAuthMessage(response.data.message || "OTP sent. Please check your email.");
    } catch (error) {
      setAuthError(error.response?.data?.message || "Unable to send OTP. Please try again.");
    } finally {
      setOtpLoading(false);
    }
  };

  const handleVerifyOtp = async () => {
    clearMessages();
    const email = registerEmail.trim().toLowerCase();

    if (!/^\d{6}$/.test(otpCode)) {
      setAuthError("Please enter the 6-digit OTP.");
      return;
    }

    setVerifyLoading(true);
    try {
      const response = await axios.post(`${API}/verify-otp`, {
        email,
        otp: otpCode,
      });
      setEmailVerified(true);
      setAuthMessage(response.data.message || "Email verified successfully.");
    } catch (error) {
      setEmailVerified(false);
      setAuthError(error.response?.data?.message || "OTP verification failed. Please try again.");
    } finally {
      setVerifyLoading(false);
    }
  };

  const handleRegister = async (event) => {
    event.preventDefault();
    clearMessages();

    if (!registerName.trim() || !registerEmail.trim() ||
        !registerBranch || !registerSemester ||
        !registerPassword || !confirmPassword) {
      setAuthError("Please fill in all fields.");
      return;
    }

    if (!emailVerified) {
      setAuthError("Please verify your email with OTP before creating your account.");
      return;
    }

    if (registerPassword.length < 6) {
      setAuthError("Password must contain at least 6 characters.");
      return;
    }

    if (registerPassword !== confirmPassword) {
      setAuthError("Password and Confirm Password do not match.");
      return;
    }

    setRegisterLoading(true);
    try {
      const response = await axios.post(`${API}/register`, {
        full_name: registerName.trim(),
        email: registerEmail.trim().toLowerCase(),
        password: registerPassword,
        college: "AKS University",
        branch: registerBranch,
        semester: Number(registerSemester),
      });

      setAuthMessage(response.data.message || "Registration successful. You can now login.");

      setRegisterName("");
      setRegisterEmail("");
      setRegisterBranch("");
      setRegisterSemester("");
      setRegisterPassword("");
      setConfirmPassword("");
      resetOtpState();

      window.setTimeout(() => {
        setShowRegister(false);
        setShowLogin(true);
        setAuthMessage("");
      }, 1200);
    } catch (error) {
      setAuthError(error.response?.data?.message || "Registration failed. Please try again.");
    } finally {
      setRegisterLoading(false);
    }
  };

  const handleLogin = async (event) => {
    event.preventDefault();
    clearMessages();

    if (!loginEmail.trim() || !loginPassword) {
      setAuthError("Please enter your email and password.");
      return;
    }

    setLoginLoading(true);
    try {
      const response = await axios.post(`${API}/login`, {
        email: loginEmail.trim().toLowerCase(),
        password: loginPassword,
      });

      setLoggedInUser(response.data.user);
      setAuthMessage(response.data.message || "Login successful!");
      setLoginPassword("");

      window.setTimeout(() => {
        setShowLogin(false);
        setAuthMessage("");
      }, 1000);
    } catch (error) {
      setAuthError(error.response?.data?.message || "Login failed. Please check your email and password.");
    } finally {
      setLoginLoading(false);
    }
  };

  // These password-reset endpoints must also be implemented in backend/app.py.
  const handleForgotSendOtp = async () => {
    clearMessages();
    const email = forgotEmail.trim().toLowerCase();

    if (!email) {
      setAuthError("Please enter the email address linked to your account.");
      return;
    }

    setForgotLoading(true);
    try {
      const response = await axios.post(`${API}/forgot-password/send-otp`, { email });
      setForgotOtpSent(true);
      setForgotOtpVerified(false);
      setAuthMessage(response.data.message || "Password reset OTP sent.");
    } catch (error) {
      setAuthError(
        error.response?.data?.message ||
        "Password reset is not connected to the backend yet. We need to add its API in app.py."
      );
    } finally {
      setForgotLoading(false);
    }
  };

  const handleForgotVerifyOtp = async () => {
    clearMessages();

    if (!/^\d{6}$/.test(forgotOtp)) {
      setAuthError("Please enter the 6-digit OTP.");
      return;
    }

    setForgotLoading(true);
    try {
      const response = await axios.post(`${API}/forgot-password/verify-otp`, {
        email: forgotEmail.trim().toLowerCase(),
        otp: forgotOtp,
      });
      setForgotOtpVerified(true);
      setAuthMessage(response.data.message || "OTP verified. Set your new password.");
    } catch (error) {
      setAuthError(error.response?.data?.message || "Could not verify the reset OTP. Check the backend API.");
    } finally {
      setForgotLoading(false);
    }
  };

  const handleResetPassword = async (event) => {
    event.preventDefault();
    clearMessages();

    if (newPassword.length < 6) {
      setAuthError("New password must contain at least 6 characters.");
      return;
    }

    if (newPassword !== confirmNewPassword) {
      setAuthError("New password and Confirm Password do not match.");
      return;
    }

    setForgotLoading(true);
    try {
      const response = await axios.post(`${API}/forgot-password/reset`, {
        email: forgotEmail.trim().toLowerCase(),
        otp: forgotOtp,
        new_password: newPassword,
      });
      setAuthMessage(response.data.message || "Password reset successful. Please login.");
      setLoginEmail(forgotEmail.trim().toLowerCase());

      window.setTimeout(() => {
        setShowForgot(false);
        setShowLogin(true);
        setAuthMessage("");
      }, 1200);
    } catch (error) {
      setAuthError(error.response?.data?.message || "Password reset failed. Check that the reset API is implemented.");
    } finally {
      setForgotLoading(false);
    }
  };

  const handleLogout = () => {
    if (!window.confirm("Are you sure you want to logout from CampusCart?")) return;
    setLoggedInUser(null);
    clearMessages();
  };

  const filteredProducts = products.filter((product) => {
    const query = searchText.trim().toLowerCase();
    if (!query) return true;

    return [
      product.name,
      product.category,
      product.branch,
      product.semester,
      product.condition,
      product.college,
    ].filter(Boolean).join(" ").toLowerCase().includes(query);
  });

  return (
    <div className="app">
      <nav className="navbar">
        <a className="logo" href="#home" aria-label="CampusCart Home">
          Campus<span>Cart</span>
        </a>

        <div className="nav-links">
          <a href="#home">Home</a>
          <a href="#explore">Explore</a>
          <a href="#how-it-works">How It Works</a>
          <a href="#cartmate">CartMate AI</a>
        </div>

        <div className="nav-actions">
          <span className="nav-campus">AKS University</span>

          {loggedInUser ? (
            <>
              <span className="nav-campus">Hi, {loggedInUser.full_name}</span>
              <button type="button" className="login-btn" onClick={handleLogout}>Logout</button>
            </>
          ) : (
            <>
              <button type="button" className="login-btn" onClick={openLogin}>Login</button>
              <button type="button" className="join-btn" onClick={openRegister}>Join Campus</button>
              <button type="button" className="sell-btn" onClick={openRegister}>Sell Product</button>
            </>
          )}
        </div>
      </nav>

      <section className="hero" id="home">
        <div className="hero-content">
          <div className="hero-badge">Verified Student Marketplace</div>
          <h1>
            Buy Smart.<br />
            Sell Easy.<br />
            <span>Reuse on Campus.</span>
          </h1>
          <p>
            CampusCart is a student-focused marketplace where verified campus
            students can buy and sell second-hand books, study material and useful college products.
          </p>

          <form
            className="hero-search"
            onSubmit={(event) => {
              event.preventDefault();
              document.getElementById("explore")?.scrollIntoView({ behavior: "smooth" });
            }}
          >
            <input
              type="search"
              value={searchText}
              onChange={(event) => setSearchText(event.target.value)}
              placeholder="Search books, CSE, semester, price..."
              aria-label="Search campus products"
            />
            <button type="submit" className="search-btn">Search</button>
          </form>
        </div>
      </section>

      <div className="campus-bar">
        <div className="campus-info">
          <div className="campus-icon">AKS</div>
          <div>
            <strong>Your Campus: AKS University, Satna</strong>
            <small>Verified campus marketplace</small>
          </div>
          <span className="verified">Verified</span>
        </div>
        <button type="button" className="change-campus" onClick={() => setAuthMessage("Campus selection is currently set to AKS University, Satna.")}>
          Change Campus
        </button>
      </div>

      <section className="section" id="explore">
        <div className="section-header">
          <div>
            <h2 className="section-title">Explore Categories</h2>
            <p className="section-subtitle">Find products made for student life</p>
          </div>
          <a href="#products" className="view-all">View Products</a>
        </div>

        <div className="category-grid">
          {categories.map((category) => (
            <button
              type="button"
              className="category-card"
              key={category.title}
              onClick={() => setSearchText(category.title)}
            >
              <div className="category-icon">{category.icon}</div>
              <h3>{category.title}</h3>
              <p>{category.description}</p>
            </button>
          ))}
        </div>
      </section>

      <section className="section" id="products">
        <div className="section-header">
          <div>
            <h2 className="section-title">Trending on Campus</h2>
            <p className="section-subtitle">Popular second-hand products from students</p>
          </div>
          <button type="button" className="view-all" onClick={() => setSearchText("")}>Clear Search</button>
        </div>

        {searchText && <p className="section-subtitle">Showing results for: <strong>{searchText}</strong></p>}

        <div className="product-grid">
          {productsLoading ? (
            <p>Loading campus products...</p>
          ) : productsError ? (
            <p>{productsError}</p>
          ) : filteredProducts.length === 0 ? (
            <p>No matching products found.</p>
          ) : (
            filteredProducts.map((product, index) => (
              <article className="product-card" key={product.product_id || index}>
                <div className="product-image">
                  <span className="condition-badge">{product.condition || "Student Listed"}</span>
                  <span className="product-image-placeholder">CampusCart</span>
                </div>
                <div className="product-content">
                  <h3>{product.name}</h3>
                  <div className="product-meta">{product.meta}</div>
                  <div className="price-row">
                    <div>
                      <span className="price">{product.displayPrice}</span>
                      {product.originalPrice && <span className="original-price">{product.originalPrice}</span>}
                    </div>
                    {product.saving && <span className="save-text">{product.saving}</span>}
                  </div>
                  <div className="product-campus">{product.campus}</div>
                  {loggedInUser ? (
                    <button type="button" className="primary-btn product-action" onClick={() => window.alert("Buyer-seller contact will be connected with the messaging module.")}>
                      Contact Seller
                    </button>
                  ) : (
                    <button type="button" className="primary-btn product-action" onClick={openLogin}>
                      Login to Connect
                    </button>
                  )}
                </div>
              </article>
            ))
          )}
        </div>
      </section>

      <section className="section">
        <div className="discovery">
          <div>
            <h2>Smart Discovery for Students</h2>
            <p>
              Search products using natural language and narrow your results
              using campus, branch, semester, subject, price and condition filters.
            </p>
            <button type="button" className="primary-btn" onClick={() => document.getElementById("products")?.scrollIntoView({ behavior: "smooth" })}>
              Explore Products
            </button>
          </div>
          <div className="discovery-box">
            <div className="discovery-search">Try searching for a DBMS book</div>
            <div className="filter-tags">
              <span className="filter-tag">CSE</span>
              <span className="filter-tag">Semester 3</span>
              <span className="filter-tag">Books</span>
              <span className="filter-tag">Under ₹400</span>
              <span className="filter-tag">Very Good</span>
            </div>
          </div>
        </div>
      </section>

      <section className="section" id="cartmate">
        <div className="cartmate">
          <div>
            <h2>Meet CartMate AI</h2>
            <p>
              Your campus marketplace assistant. CartMate is planned to help
              students discover products, find relevant books and get marketplace
              assistance in English, Hindi and Hinglish.
            </p>
            <div className="ai-features">
              <span className="ai-feature">Product Search</span>
              <span className="ai-feature">Book Discovery</span>
              <span className="ai-feature">Recommendations</span>
              <span className="ai-feature">Hindi / Hinglish</span>
            </div>
          </div>
          <div className="ai-panel">
            <div className="ai-message user">CSE 3rd sem ki DBMS book ₹400 ke under chahiye.</div>
            <div className="ai-message">I can help you find relevant DBMS books and compare their listed prices.</div>
            <div className="ai-message user">Very good condition wali dikhao.</div>
          </div>
        </div>
      </section>

      <section className="section" id="how-it-works">
        <div className="section-header">
          <div>
            <h2 className="section-title">How CampusCart Works</h2>
            <p className="section-subtitle">Simple, campus-focused buying and selling</p>
          </div>
        </div>
        <div className="steps-grid">
          {[
            ["1", "Verify", "Create your student account and verify your email."],
            ["2", "List", "Add your second-hand books or student products."],
            ["3", "Discover", "Search products available on campus."],
            ["4", "Connect", "Contact the seller to discuss the product."],
            ["5", "Exchange", "Complete the student-to-student exchange."],
          ].map(([number, title, description]) => (
            <div className="step-card" key={number}>
              <div className="step-number">{number}</div>
              <h3>{title}</h3>
              <p>{description}</p>
            </div>
          ))}
        </div>
      </section>

      <section className="section">
        <div className="network">
          <h2>Built for a Growing Campus Network</h2>
          <p>
            CampusCart starts with AKS University and is designed to support
            verified campus communities in the future.
          </p>
          <div className="network-cards">
            <div className="network-card">AKS University</div>
            <div className="network-card">Your Campus</div>
            <div className="network-card">Verified Students</div>
            <div className="network-card">Future Campus Network</div>
          </div>
        </div>
      </section>

      <footer className="footer">
        <div className="footer-main">
          <div>
            <div className="footer-logo">CampusCart</div>
            <p className="footer-description">
              A student marketplace designed to buy, sell and reuse useful products.
            </p>
          </div>
          <div className="footer-column">
            <h4>Marketplace</h4>
            <a href="#products">Browse Products</a>
            <a href="#explore">Categories</a>
            <button type="button" onClick={openRegister}>Sell Product</button>
          </div>
          <div className="footer-column">
            <h4>Campus</h4>
            <a href="#home">Student Verification</a>
            <a href="#how-it-works">How It Works</a>
          </div>
          <div className="footer-column">
            <h4>Support</h4>
            <a href="#cartmate">CartMate AI</a>
            <button type="button" onClick={openForgot}>Forgot Password?</button>
          </div>
        </div>
        <div className="footer-bottom">
          <span>© 2026 CampusCart. Student Marketplace.</span>
          <span>Built for campus communities</span>
        </div>
      </footer>

      {authMessage && !showLogin && !showRegister && !showForgot && (
        <div className="global-message auth-success">{authMessage}</div>
      )}

      {showLogin && (
        <div className="modal-overlay" onClick={closeModals}>
          <section className="modal" role="dialog" aria-modal="true" aria-labelledby="login-heading" onClick={(event) => event.stopPropagation()}>
            <button type="button" className="modal-close" onClick={closeModals} aria-label="Close">×</button>
            <div className="auth-brand-mark">CC</div>
            <p className="auth-eyebrow">AKS UNIVERSITY • STUDENT MARKETPLACE</p>
            <h2 id="login-heading">Welcome Back</h2>
            <p className="modal-subtitle">Sign in to continue to your CampusCart account.</p>

            {authError && <div className="auth-error" role="alert">{authError}</div>}
            {authMessage && <div className="auth-success" role="status">{authMessage}</div>}

            <form onSubmit={handleLogin}>
              <div className="form-group">
                <label htmlFor="login-email">College Email</label>
                <input id="login-email" type="email" placeholder="Enter your email address" value={loginEmail} onChange={(event) => setLoginEmail(event.target.value)} autoComplete="email" required />
              </div>

              <PasswordField
                label="Password"
                value={loginPassword}
                onChange={(event) => setLoginPassword(event.target.value)}
                placeholder="Enter your password"
                autoComplete="current-password"
              />

              <div className="forgot-row">
                <button type="button" className="text-link" onClick={openForgot}>Forgot Password?</button>
              </div>

              <button type="submit" className="modal-submit" disabled={loginLoading}>
                {loginLoading ? "Signing In..." : "Sign In to CampusCart"}
              </button>
            </form>

            <div className="auth-divider"><span>NEW TO CAMPUSCART?</span></div>
            <p className="switch-auth">Don't have an account? <button type="button" onClick={openRegister}>Join Campus</button></p>
          </section>
        </div>
      )}

      {showRegister && (
        <div className="modal-overlay" onClick={closeModals}>
          <section className="modal register-modal" role="dialog" aria-modal="true" aria-labelledby="register-heading" onClick={(event) => event.stopPropagation()}>
            <button type="button" className="modal-close" onClick={closeModals} aria-label="Close">×</button>
            <div className="auth-brand-mark">CC</div>
            <p className="auth-eyebrow">YOUR CAMPUS. YOUR COMMUNITY.</p>
            <h2 id="register-heading">Join CampusCart</h2>
            <p className="modal-subtitle">Create your verified student marketplace account.</p>

            <div className="verification-note">Verify your email with OTP before creating your student account.</div>
            {authError && <div className="auth-error" role="alert">{authError}</div>}
            {authMessage && <div className="auth-success" role="status">{authMessage}</div>}

            <form onSubmit={handleRegister}>
              <div className="form-group">
                <label htmlFor="register-name">Full Name</label>
                <input id="register-name" type="text" placeholder="Enter your full name" value={registerName} onChange={(event) => setRegisterName(event.target.value)} autoComplete="name" required />
              </div>

              <div className="form-group">
                <label htmlFor="register-email">College Email</label>
                <input id="register-email" type="email" placeholder="Enter your email address" value={registerEmail} onChange={handleRegisterEmailChange} autoComplete="email" required />
              </div>

              {!emailVerified && (
                <button type="button" className="secondary-submit" onClick={handleSendOtp} disabled={otpLoading}>
                  {otpLoading ? "Sending OTP..." : otpSent ? "Resend Verification OTP" : "Send Verification OTP"}
                </button>
              )}

              {otpSent && !emailVerified && (
                <div className="form-group otp-group">
                  <label htmlFor="register-otp">6-Digit Email OTP</label>
                  <input id="register-otp" type="text" inputMode="numeric" maxLength={6} placeholder="Enter OTP received by email" value={otpCode} onChange={(event) => setOtpCode(event.target.value.replace(/\D/g, ""))} required />
                  <button type="button" className="secondary-submit" onClick={handleVerifyOtp} disabled={verifyLoading}>
                    {verifyLoading ? "Verifying..." : "Verify Email"}
                  </button>
                </div>
              )}

              {emailVerified && <div className="auth-success">✓ Email verified successfully.</div>}

              <div className="form-row">
                <div className="form-group">
                  <label htmlFor="register-branch">Branch</label>
                  <select id="register-branch" value={registerBranch} onChange={(event) => setRegisterBranch(event.target.value)} required>
                    <option value="">Select Branch</option>
                    <option value="CSE">CSE</option>
                    <option value="AI & ML">AI &amp; ML</option>
                    <option value="IT">IT</option>
                    <option value="ECE">ECE</option>
                    <option value="Mechanical">Mechanical</option>
                    <option value="Other">Other</option>
                  </select>
                </div>

                <div className="form-group">
                  <label htmlFor="register-semester">Semester</label>
                  <select id="register-semester" value={registerSemester} onChange={(event) => setRegisterSemester(event.target.value)} required>
                    <option value="">Select Semester</option>
                    {[1, 2, 3, 4, 5, 6, 7, 8].map((semester) => (
                      <option value={String(semester)} key={semester}>Semester {semester}</option>
                    ))}
                  </select>
                </div>
              </div>

              <PasswordField
                label="Create Password"
                value={registerPassword}
                onChange={(event) => setRegisterPassword(event.target.value)}
                placeholder="At least 6 characters"
                autoComplete="new-password"
              />

              <PasswordField
                label="Confirm Password"
                value={confirmPassword}
                onChange={(event) => setConfirmPassword(event.target.value)}
                placeholder="Re-enter your password"
                autoComplete="new-password"
              />

              <button type="submit" className="modal-submit" disabled={!emailVerified || registerLoading}>
                {registerLoading ? "Creating Account..." : "Create Student Account"}
              </button>
            </form>

            <p className="switch-auth">Already have an account? <button type="button" onClick={openLogin}>Login</button></p>
          </section>
        </div>
      )}

      {showForgot && (
        <div className="modal-overlay" onClick={closeModals}>
          <section className="modal" role="dialog" aria-modal="true" aria-labelledby="forgot-heading" onClick={(event) => event.stopPropagation()}>
            <button type="button" className="modal-close" onClick={closeModals} aria-label="Close">×</button>
            <div className="auth-brand-mark">CC</div>
            <p className="auth-eyebrow">ACCOUNT RECOVERY</p>
            <h2 id="forgot-heading">Forgot Password?</h2>
            <p className="modal-subtitle">Recover access to your CampusCart student account.</p>

            {authError && <div className="auth-error" role="alert">{authError}</div>}
            {authMessage && <div className="auth-success" role="status">{authMessage}</div>}

            <form onSubmit={handleResetPassword}>
              <div className="form-group">
                <label htmlFor="forgot-email">Registered Email</label>
                <input id="forgot-email" type="email" placeholder="Enter your registered email" value={forgotEmail} onChange={(event) => {
                  setForgotEmail(event.target.value);
                  setForgotOtpSent(false);
                  setForgotOtpVerified(false);
                  setForgotOtp("");
                  clearMessages();
                }} autoComplete="email" required />
              </div>

              {!forgotOtpVerified && (
                <button type="button" className="secondary-submit" onClick={handleForgotSendOtp} disabled={forgotLoading}>
                  {forgotLoading ? "Please Wait..." : forgotOtpSent ? "Resend Reset OTP" : "Send Reset OTP"}
                </button>
              )}

              {forgotOtpSent && !forgotOtpVerified && (
                <div className="form-group">
                  <label htmlFor="forgot-otp">Reset OTP</label>
                  <input id="forgot-otp" type="text" inputMode="numeric" maxLength={6} placeholder="Enter 6-digit OTP" value={forgotOtp} onChange={(event) => setForgotOtp(event.target.value.replace(/\D/g, ""))} required />
                  <button type="button" className="secondary-submit" onClick={handleForgotVerifyOtp} disabled={forgotLoading}>
                    {forgotLoading ? "Verifying..." : "Verify Reset OTP"}
                  </button>
                </div>
              )}

              {forgotOtpVerified && (
                <>
                  <PasswordField label="New Password" value={newPassword} onChange={(event) => setNewPassword(event.target.value)} placeholder="At least 6 characters" autoComplete="new-password" />
                  <PasswordField label="Confirm New Password" value={confirmNewPassword} onChange={(event) => setConfirmNewPassword(event.target.value)} placeholder="Re-enter new password" autoComplete="new-password" />
                  <button type="submit" className="modal-submit" disabled={forgotLoading}>
                    {forgotLoading ? "Resetting Password..." : "Reset Password"}
                  </button>
                </>
              )}
            </form>

            <p className="switch-auth"><button type="button" onClick={openLogin}>← Back to Login</button></p>
          </section>
        </div>
      )}
    </div>
  );
}

export default App;
